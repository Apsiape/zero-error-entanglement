\subsection*{Notation and glossary}
\addcontentsline{toc}{subsection}{Notation and glossary}
\begin{center}\small
\begin{tabular}{@{}>{\raggedright\arraybackslash}p{.26\linewidth}>{\raggedright\arraybackslash}p{.67\linewidth}@{}}\toprule
symbol or term&meaning\\\midrule
$N,m$&finite rational channel and number of messages\\
$G,\Gcat;\ \Gamma(G)$&a game, the Catalan assembly game; its tagged confusability graph\\
$k;\ \mathcal C_c$&number of tagged local questions; a clique in their partition\\
$n,\lambda=2^{-n}$&number of FMS bridges and creation covariance ratio\\
$\drow,\qclk$&finite row-ancilla dimension and clock order/dimension\\
$D,h=D^2$&positive Schmidt operator and pure-game marginal\\
$\rho$&channel-code marginal, equal to $D/\Tr D$ in the forward map\\
$h_0,h_{F,\lambda},\sigma$&source density, four-letter tree density and spectator density\\
$\hcat,\kcat$&critical Catalan density and its normalized square\\
$\Ccat$&Catalan height counter; $C_{\rm ctr}$ is the computation counter\\
$\Omega_n,\mu_n,\nu_n$&scaffold cover constant, scaffold density weights and their normalized square roots\\
$\WW,\WW_{\rm Cat};\ u(w)$&four-letter words, legal Catalan words; number of Catalan up-steps\\
$M_g;\ \mathrm{Mat}_r$&magic support of bridge $g$; scalar $r$-by-$r$ matrix algebra\\
$\tau$&tracial state on the commuting centralizer\\\bottomrule
\end{tabular}
\end{center}
\noindent\emph{Normal}: represented by a trace-one positive trace-class operator.
\emph{Centralizer}: operators commuting with a density, or satisfying $\phi(XT)=\phi(TX)$ for a state $\phi$.
\emph{Bridge}: an FMS test fixing the density ratio between two selected corners.
\emph{Scaffold}: the finite corner system joining those bridges.
\emph{Summable Schmidt amplitudes}: $\Tr D<\infty$.
\emph{Saturating code}: $m=k$ for the specified clique partition.
\emph{YES/NO}: an instance with/without the specified perfect strategy or code.
